### 问题3：动爻计算可能不符合标准 ⚠️

**位置**: `src/meihua/qigua.ts` 第 57-60 行

**当前实现**:
```typescript
function getChangingLine(num: number): number {
  const line = ((num - 1) % 6) + 1;  // ⚠️ 需验证
  return line;
}
```

**问题分析**:
- 公式看起来正确：`(n-1) % 6 + 1` 确实能保证结果在1-6范围
- ⚠️ 但需要确认：当余数为0时是否应该是6？

**正确做法**（根据权威资料）:
> "以年月日时总和除以六，以余数为动爻"
> "若余数为0，则取为6"

**验证**:
- 如果 num = 6: (6-1) % 6 + 1 = 5 % 6 + 1 = 5 + 1 = 6 ✅
- 如果 num = 12: (12-1) % 6 + 1 = 11 % 6 + 1 = 5 + 1 = 6 ✅
- 如果 num = 7: (7-1) % 6 + 1 = 6 % 6 + 1 = 0 + 1 = 1 ✅

**结论**: ✅ 这个实现是正确的

### 方案1：修复变卦计算

```typescript
// 八卦到二进制映射（阳=1，阴=0，从下往上）
const GUA_TO_BINARY: Record<BaGua, string> = {
  '乾': '111',  // ☰ 三阳爻
  '兑': '110',  // ☱ 上阳下阴
  '离': '101',  // ☲ 中阴
  '震': '100',  // ☳ 下阳
  '巽': '011',  // ☴ 下阴
  '坎': '010',  // ☵ 中阳
  '艮': '001',  // ☶ 上阴下阳
  '坤': '000'   // ☷ 三阴爻
};

function getChangeGua(upperGua: BaGua, lowerGua: BaGua, changingLine: number): { upperGua: BaGua; lowerGua: BaGua } {
  // 1. 组合成六爻
  const lowerBinary = GUA_TO_BINARY[lowerGua];  // 1-3爻
  const upperBinary = GUA_TO_BINARY[upperGua];  // 4-6爻
  const sixLines = lowerBinary + upperBinary;    // 组合
  
  // 2. 翻转动爻（从下往上数，所以是 changingLine-1）
  const lines = sixLines.split('');
  const index = changingLine - 1;
  lines[index] = lines[index] === '1' ? '0' : '1';  // 阳变阴，阴变阳
  const newSixLines = lines.join('');
  
  // 3. 拆分回上下卦
  const newLowerBinary = newSixLines.substring(0, 3);
  const newUpperBinary = newSixLines.substring(3, 6);
  
  // 4. 转回八卦
  const binaryToGua = Object.fromEntries(
    Object.entries(GUA_TO_BINARY).map(([k, v]) => [v, k as BaGua])
  );
  
  return {
    lowerGua: binaryToGua[newLowerBinary],
    upperGua: binaryToGua[newUpperBinary]
  };
}
```
